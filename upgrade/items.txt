--- 1.2  Fix pcode slash → hyphen (manufactured items) -----------------------

-- Preview:
SELECT id, pcode FROM items WHERE pcode LIKE '%/%' AND deleted_at IS NULL LIMIT 50;

UPDATE items
SET pcode = REPLACE(pcode, '/', '-')
WHERE pcode LIKE '%/%' AND deleted_at IS NULL;

UPDATE item_groups
SET name = REPLACE(name, '/', '-')
WHERE name LIKE '%/%';


--- 1.3  Copy item_groups.alias → item_groups.name (BEFORE alias column drop) --

-- Preview groups where alias has a real product name but name does not:
SELECT id, name, alias, master, variant
FROM item_groups
WHERE alias IS NOT NULL
  AND TRIM(alias) <> ''
  AND (
    name IS NULL
    OR TRIM(name) = ''
    OR UPPER(name) = UPPER(alias)
  )
LIMIT 50;

-- Promote alias to name where name is empty:
UPDATE item_groups
SET name = UPPER(TRIM(alias))
WHERE alias IS NOT NULL
  AND TRIM(alias) <> ''
  AND (name IS NULL OR TRIM(name) = '');

-- Where name equals pcode placeholder but alias has the real product name:
UPDATE item_groups g
JOIN (
    SELECT group_id, UPPER(TRIM(pcode)) AS pcode
    FROM items
    WHERE deleted_at IS NULL
    GROUP BY group_id, UPPER(TRIM(pcode))
) i ON i.group_id = g.id
SET g.name = UPPER(TRIM(g.alias))
WHERE g.alias IS NOT NULL
  AND TRIM(g.alias) <> ''
  AND UPPER(TRIM(g.name)) = i.pcode
  AND UPPER(TRIM(g.alias)) <> i.pcode;

--------------------------------------------------------------------------------
REFERENCE — NEW IDENTITY RULES (after migration)
--------------------------------------------------------------------------------

SKU (items.code):
  Manufactured:  {TYPE}-{pcode}-{size?}     e.g. AJD-CX90324-05-S
  Asset lancar:  {pcode}-{warna}-{size?}    e.g. GLOVE-01-BLACK-S
  All-size (AS): omit trailing -{size}

Display name (items.name):
  {group.name} - {warna code} - {size code}
  e.g. SLASH RUNNING SHIRT - BLUE - S
  or   CX93249-03 - BLUE - S  (when product name not yet released)

Pcode:
  Manufactured:  XX12345-23  (hyphen, not slash)
  Asset lancar:  GLOVE-01

Product name (item_groups.name):
  Required for asset lancar.
  Optional for manufactured items — defaults to pcode until renamed via
  item edit or group detail page.
